# Bits to Banking | Umicom Foundation | C23 native learning exercise
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingArena LANGUAGES C)
include(CTest)
add_library(btb_arena STATIC arena.c)
target_include_directories(btb_arena PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bits2banking-arena-demo main.c)
target_link_libraries(bits2banking-arena-demo PRIVATE btb_arena)
add_executable(bits2banking-arena-test test_arena.c)
target_link_libraries(bits2banking-arena-test PRIVATE btb_arena)
foreach(target IN ITEMS btb_arena bits2banking-arena-demo bits2banking-arena-test)
    set_target_properties("${target}" PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options("${target}" PRIVATE /W4)
    else()
        target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.arena.ownership COMMAND bits2banking-arena-test)
    add_test(NAME bits2banking.arena.report COMMAND bits2banking-arena-demo)
    set_tests_properties(bits2banking.arena.ownership bits2banking.arena.report PROPERTIES LABELS "learning;c23;memory")
endif()
