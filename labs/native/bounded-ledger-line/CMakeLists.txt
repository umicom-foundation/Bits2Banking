# Bits to Banking | Umicom Foundation | C23 native learning exercise
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingLedgerLine LANGUAGES C)
include(CTest)
add_library(btb_ledger_line STATIC ledger_line.c)
target_include_directories(btb_ledger_line PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bits2banking-ledger-line-demo main.c)
target_link_libraries(bits2banking-ledger-line-demo PRIVATE btb_ledger_line)
add_executable(bits2banking-ledger-line-test test_ledger_line.c)
target_link_libraries(bits2banking-ledger-line-test PRIVATE btb_ledger_line)
add_executable(bits2banking-ledger-line-fuzz fuzz_ledger_line.c)
target_link_libraries(bits2banking-ledger-line-fuzz PRIVATE btb_ledger_line)
foreach(target IN ITEMS btb_ledger_line bits2banking-ledger-line-demo
                        bits2banking-ledger-line-test bits2banking-ledger-line-fuzz)
    set_target_properties("${target}" PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options("${target}" PRIVATE /W4)
    else()
        target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.ledger_line.syntax COMMAND bits2banking-ledger-line-test)
    add_test(NAME bits2banking.ledger_line.mutations COMMAND bits2banking-ledger-line-fuzz)
    add_test(NAME bits2banking.ledger_line.example COMMAND bits2banking-ledger-line-demo)
    set_tests_properties(bits2banking.ledger_line.syntax bits2banking.ledger_line.mutations
        bits2banking.ledger_line.example PROPERTIES LABELS "learning;c23;bytes;fuzz")
endif()
