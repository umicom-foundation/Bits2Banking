# Bits2Banking native C23 web-service lesson | Sammy Hegab, Umicom Foundation | MIT
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingHttpLoopback LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb_http STATIC http.c static_files.c route_service.c response_progress.c learning_routes.c
    cache_validator.c admission_queue.c)
target_include_directories(btb_http PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-http-server main.c tcp_server.c)
target_link_libraries(btb-http-server PRIVATE btb_http)
if(WIN32)
  target_link_libraries(btb-http-server PRIVATE ws2_32)
endif()
add_executable(btb-http-test test_http.c)
add_executable(btb-http-static-test test_static.c)
add_executable(btb-http-route-test test_routes.c)
add_executable(btb-http-progress-test test_progress.c)
target_link_libraries(btb-http-test PRIVATE btb_http)
target_link_libraries(btb-http-static-test PRIVATE btb_http)
target_link_libraries(btb-http-route-test PRIVATE btb_http)
target_link_libraries(btb-http-progress-test PRIVATE btb_http)
add_executable(btb-http-cache-test test_cache.c)
add_executable(btb-http-admission-test test_admission.c)
add_executable(btb-admission-demo admission_demo.c)
target_link_libraries(btb-http-cache-test PRIVATE btb_http)
target_link_libraries(btb-http-admission-test PRIVATE btb_http)
target_link_libraries(btb-admission-demo PRIVATE btb_http)
foreach(target IN ITEMS btb_http btb-http-server btb-http-test btb-http-static-test btb-http-route-test btb-http-progress-test btb-http-cache-test btb-http-admission-test btb-admission-demo)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endforeach()
if(BUILD_TESTING)
  add_test(NAME bits2banking.http.parse COMMAND btb-http-test parsing)
  add_test(NAME bits2banking.http.responses COMMAND btb-http-test responses)
  add_test(NAME bits2banking.http.mutations COMMAND btb-http-test mutations)
  add_test(NAME bits2banking.http.static_parsing COMMAND btb-http-static-test parsing)
  add_test(NAME bits2banking.http.static_files COMMAND btb-http-static-test files)
  add_test(NAME bits2banking.http.routing COMMAND btb-http-route-test routing)
  add_test(NAME bits2banking.http.service_lifetime COMMAND btb-http-route-test lifetime)
  add_test(NAME bits2banking.http.route_mutations COMMAND btb-http-route-test mutations)
  add_test(NAME bits2banking.http.response_completion COMMAND btb-http-progress-test completion)
  add_test(NAME bits2banking.http.disconnect_semantics COMMAND btb-http-progress-test disconnect)
  add_test(NAME bits2banking.http.help COMMAND btb-http-server --help)
  set_tests_properties(bits2banking.http.help PROPERTIES PASS_REGULAR_EXPRESSION "127.0.0.1")
  add_test(NAME bits2banking.http.cache.validators COMMAND btb-http-cache-test validators)
  add_test(NAME bits2banking.http.cache.files COMMAND btb-http-cache-test files
    "${CMAKE_CURRENT_SOURCE_DIR}/sample_site")
  add_test(NAME bits2banking.http.admission.queue COMMAND btb-http-admission-test queue)
  add_test(NAME bits2banking.http.admission.mutations COMMAND btb-http-admission-test mutations)
  add_test(NAME bits2banking.http.admission.demo COMMAND btb-admission-demo)
endif()
