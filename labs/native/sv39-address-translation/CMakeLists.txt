# Bits2Banking — three-level Sv39 page-table laboratory, C23 only.
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingSv39 LANGUAGES C)
include(CTest)
add_library(btb_sv39 STATIC sv39.c)
set_target_properties(btb_sv39 PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(btb_sv39 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-sv39-demo main.c)
target_link_libraries(btb-sv39-demo PRIVATE btb_sv39)
add_executable(btb-sv39-test test_sv39.c)
target_link_libraries(btb-sv39-test PRIVATE btb_sv39)
foreach(target IN ITEMS btb_sv39 btb-sv39-demo btb-sv39-test)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.sv39.walk COMMAND btb-sv39-test)
    add_test(NAME bits2banking.sv39.demo COMMAND btb-sv39-demo)
    set_tests_properties(bits2banking.sv39.walk bits2banking.sv39.demo
                         PROPERTIES LABELS "education;c23;sv39;virtual-memory")
endif()
