# Bits2Banking | Umicom Foundation | ISO C23 administration-evidence laboratory
# This isolated teaching target never inspects or modifies host services.
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingSysadminEvidence LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb-sysadmin-evidence-lib STATIC admin_evidence.c)
target_include_directories(btb-sysadmin-evidence-lib PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-sysadmin-evidence-demo main.c)
add_executable(btb-sysadmin-evidence-test test_evidence.c)
target_link_libraries(btb-sysadmin-evidence-demo PRIVATE btb-sysadmin-evidence-lib)
target_link_libraries(btb-sysadmin-evidence-test PRIVATE btb-sysadmin-evidence-lib)
foreach(target IN ITEMS btb-sysadmin-evidence-lib btb-sysadmin-evidence-demo btb-sysadmin-evidence-test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endforeach()
if(BUILD_TESTING)
    foreach(case_name IN ITEMS ready warning optional-fail optional-notrun missing failure notrun
            required-warning stale duplicate full invalid preserve capture repeated)
        add_test(NAME "bits2banking.sysadmin-evidence.${case_name}"
                 COMMAND btb-sysadmin-evidence-test "${case_name}")
    endforeach()
    add_test(NAME bits2banking.sysadmin-evidence.demo COMMAND btb-sysadmin-evidence-demo)
endif()
