# Bits2Banking | Umicom Foundation | Native C23-only work handoff laboratory
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingWorkHandoff LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb-work-handoff STATIC handoff.c)
target_include_directories(btb-work-handoff PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-work-handoff-test test_handoff.c)
add_executable(btb-work-handoff-demo handoff_demo.c)
target_link_libraries(btb-work-handoff-test PRIVATE btb-work-handoff)
target_link_libraries(btb-work-handoff-demo PRIVATE btb-work-handoff)
foreach(target IN ITEMS btb-work-handoff btb-work-handoff-test btb-work-handoff-demo)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endforeach()
if(BUILD_TESTING)
  add_test(NAME bits2banking.handoff.fifo COMMAND btb-work-handoff-test fifo)
  add_test(NAME bits2banking.handoff.cancel COMMAND btb-work-handoff-test cancel)
  add_test(NAME bits2banking.handoff.generation COMMAND btb-work-handoff-test generation)
  add_test(NAME bits2banking.handoff.mutations COMMAND btb-work-handoff-test mutations)
  add_test(NAME bits2banking.handoff.demo COMMAND btb-work-handoff-demo)
endif()
