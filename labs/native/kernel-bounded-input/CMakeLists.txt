# Compile the real freestanding parser as a hosted lesson, without hardware entry.
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingKernelInput LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
set(UMICOM_KERNEL_SOURCE "" CACHE PATH "Umicom Kernel source checkout")
if(NOT EXISTS "${UMICOM_KERNEL_SOURCE}/kernel/console_line.c" OR
   NOT EXISTS "${UMICOM_KERNEL_SOURCE}/include/umicom/kernel/console_shell.h")
    message(FATAL_ERROR "Set UMICOM_KERNEL_SOURCE to the Umicom Kernel checkout")
endif()
add_library(btb-kernel-input STATIC "${UMICOM_KERNEL_SOURCE}/kernel/console_line.c")
target_include_directories(btb-kernel-input PUBLIC "${UMICOM_KERNEL_SOURCE}/include")
add_executable(btb-kernel-input-demo main.c)
target_link_libraries(btb-kernel-input-demo PRIVATE btb-kernel-input)
if(BUILD_TESTING)
    add_executable(btb-kernel-input-tests test_input.c)
    target_link_libraries(btb-kernel-input-tests PRIVATE btb-kernel-input)
    foreach(_case IN ITEMS span rejected unchanged line-loss)
        add_test(NAME "bits2banking.kernel_input.${_case}" COMMAND btb-kernel-input-tests "${_case}")
        set_tests_properties("bits2banking.kernel_input.${_case}" PROPERTIES TIMEOUT 10)
    endforeach()
endif()
foreach(_target IN ITEMS btb-kernel-input btb-kernel-input-demo btb-kernel-input-tests)
    if(TARGET ${_target} AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
