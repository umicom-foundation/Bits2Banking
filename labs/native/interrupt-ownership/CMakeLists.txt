# Bits2Banking | Umicom Foundation | C23 interrupt ownership model | MIT
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(Bits2BankingInterruptOwner LANGUAGES C)
    include(CTest)
endif()
add_library(btb_interrupt_owner STATIC owner.c)
target_include_directories(btb_interrupt_owner PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(btb_interrupt_owner PROPERTIES
    C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(btb_interrupt_owner PRIVATE /W4)
else()
    target_compile_options(btb_interrupt_owner PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(btb-interrupt-ownership main.c)
    add_executable(btb-interrupt-ownership-test test_owner.c)
    foreach(target IN ITEMS btb-interrupt-ownership btb-interrupt-ownership-test)
        target_link_libraries(${target} PRIVATE btb_interrupt_owner)
        set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    endforeach()
    if(BUILD_TESTING)
        add_test(NAME bits2banking.interrupt_owner.demo COMMAND btb-interrupt-ownership)
        add_test(NAME bits2banking.interrupt_owner.contracts COMMAND btb-interrupt-ownership-test)
        set_tests_properties(bits2banking.interrupt_owner.demo bits2banking.interrupt_owner.contracts
            PROPERTIES LABELS "education;interrupt;ownership;c23")
    endif()
endif()
