cmake_minimum_required(VERSION 3.24)
project(Bits2BankingFat16 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb_fat16 STATIC fat16.c)
if(MSVC)
 target_compile_options(btb_fat16 PRIVATE /W4)
else()
 target_compile_options(btb_fat16 PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(btb-fat16-inspect main.c)
target_link_libraries(btb-fat16-inspect PRIVATE btb_fat16)
add_executable(btb-fat16-test test_fat16.c)
target_link_libraries(btb-fat16-test PRIVATE btb_fat16)
if(BUILD_TESTING)
 add_test(NAME bits2banking.fat16.parser COMMAND btb-fat16-test)
endif()
