cmake_minimum_required(VERSION 3.24)
project(Bits2BankingAllocationFailures LANGUAGES C)
include(CTest)
add_library(btb_fault_note STATIC note.c)
target_include_directories(btb_fault_note PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bits2banking-fault-injection main.c)
target_link_libraries(bits2banking-fault-injection PRIVATE btb_fault_note)
add_executable(bits2banking-fault-injection-test test_note.c)
target_link_libraries(bits2banking-fault-injection-test PRIVATE btb_fault_note)
foreach(target IN ITEMS btb_fault_note bits2banking-fault-injection bits2banking-fault-injection-test)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.fault_injection COMMAND bits2banking-fault-injection-test)
    set_tests_properties(bits2banking.fault_injection PROPERTIES LABELS "c23;education;ownership;allocators")
endif()
