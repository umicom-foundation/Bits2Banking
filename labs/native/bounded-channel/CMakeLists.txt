# Bits to Banking | Umicom Foundation | C23 copied-message classroom laboratory
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingBoundedChannel LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(btb_bounded_channel STATIC channel.c)
target_include_directories(btb_bounded_channel PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-bounded-channel main.c)
target_link_libraries(btb-bounded-channel PRIVATE btb_bounded_channel)
if(BUILD_TESTING)
    add_executable(btb-bounded-channel-test test_channel.c)
    target_link_libraries(btb-bounded-channel-test PRIVATE btb_bounded_channel)
    add_test(NAME bits2banking.bounded_channel COMMAND btb-bounded-channel-test)
    add_test(NAME bits2banking.bounded_channel.demo COMMAND btb-bounded-channel)
    set_tests_properties(bits2banking.bounded_channel.demo PROPERTIES
        PASS_REGULAR_EXPRESSION "Notes: save the weekly report")
endif()
foreach(target IN ITEMS btb_bounded_channel btb-bounded-channel)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(TARGET btb-bounded-channel-test AND NOT MSVC)
    target_compile_options(btb-bounded-channel-test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
