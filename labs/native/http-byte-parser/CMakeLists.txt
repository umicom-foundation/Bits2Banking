cmake_minimum_required(VERSION 3.24)
project(Bits2BankingHttpByteParser LANGUAGES C)
include(CTest)
add_library(btb_request_line STATIC request_line.c)
target_include_directories(btb_request_line PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bits2banking-http-parser main.c)
target_link_libraries(bits2banking-http-parser PRIVATE btb_request_line)
add_executable(bits2banking-http-parser-test test_request_line.c)
target_link_libraries(bits2banking-http-parser-test PRIVATE btb_request_line)
add_executable(bits2banking-http-seeded-fuzz fuzz_request_line.c)
target_compile_definitions(bits2banking-http-seeded-fuzz PRIVATE BTB_SEEDED_FUZZ_DRIVER)
target_link_libraries(bits2banking-http-seeded-fuzz PRIVATE btb_request_line)
foreach(target IN ITEMS btb_request_line bits2banking-http-parser bits2banking-http-parser-test bits2banking-http-seeded-fuzz)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.http_parser COMMAND bits2banking-http-parser-test)
    add_test(NAME bits2banking.http_seeded_fuzz COMMAND bits2banking-http-seeded-fuzz)
    set_tests_properties(bits2banking.http_parser bits2banking.http_seeded_fuzz
        PROPERTIES LABELS "c23;education;http;bounded-input")
endif()
