cmake_minimum_required(VERSION 3.24)
project(Bits2BankingProcessLifecycle LANGUAGES C)
include(CTest)
add_library(btb_process STATIC process.c)
target_compile_features(btb_process PUBLIC c_std_23)
target_include_directories(btb_process PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-process-demo main.c)
add_executable(btb-process-test test_process.c)
target_link_libraries(btb-process-demo PRIVATE btb_process)
target_link_libraries(btb-process-test PRIVATE btb_process)
foreach(target IN ITEMS btb_process btb-process-demo btb-process-test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.process.lifecycle COMMAND btb-process-test lifecycle)
    add_test(NAME bits2banking.process.refusals COMMAND btb-process-test refusals)
    set_tests_properties(bits2banking.process.lifecycle bits2banking.process.refusals
        PROPERTIES LABELS "bits2banking;c23;kernel-learning;state-machines")
endif()
