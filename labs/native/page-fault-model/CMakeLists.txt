cmake_minimum_required(VERSION 3.24)
project(Bits2BankingPageFaultModel LANGUAGES C)
include(CTest)
add_library(btb_page_map STATIC page_map.c)
target_compile_features(btb_page_map PUBLIC c_std_23)
target_include_directories(btb_page_map PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-page-fault-demo main.c)
add_executable(btb-page-fault-test test_page_map.c)
target_link_libraries(btb-page-fault-demo PRIVATE btb_page_map)
target_link_libraries(btb-page-fault-test PRIVATE btb_page_map)
foreach(target IN ITEMS btb_page_map btb-page-fault-demo btb-page-fault-test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.page_fault_model.basic COMMAND btb-page-fault-test basic)
    add_test(NAME bits2banking.page_fault_model.boundaries COMMAND btb-page-fault-test boundaries)
    set_tests_properties(bits2banking.page_fault_model.basic
        bits2banking.page_fault_model.boundaries
        PROPERTIES LABELS "bits2banking;c23;memory;kernel-learning")
endif()
