# Bits2Banking | Umicom Foundation | C23 bounded diagnostic ring | MIT
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingDiagnosticRing LANGUAGES C)
include(CTest)
add_library(btb_diagnostic_ring STATIC ring.c)
target_include_directories(btb_diagnostic_ring PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-diagnostic-ring main.c)
add_executable(btb-diagnostic-ring-test test_ring.c)
foreach(target IN ITEMS btb_diagnostic_ring btb-diagnostic-ring btb-diagnostic-ring-test)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
target_link_libraries(btb-diagnostic-ring PRIVATE btb_diagnostic_ring)
target_link_libraries(btb-diagnostic-ring-test PRIVATE btb_diagnostic_ring)
if(BUILD_TESTING)
    add_test(NAME bits2banking.diagnostic_ring.demo COMMAND btb-diagnostic-ring)
    add_test(NAME bits2banking.diagnostic_ring.contracts COMMAND btb-diagnostic-ring-test)
    set_tests_properties(bits2banking.diagnostic_ring.demo bits2banking.diagnostic_ring.contracts
        PROPERTIES LABELS "education;c23;diagnostics;retention")
endif()
