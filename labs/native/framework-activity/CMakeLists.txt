cmake_minimum_required(VERSION 3.24)
project(BtbFrameworkActivity LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
find_package(UmicomFramework CONFIG REQUIRED COMPONENTS base)
option(BTB_ACTIVITY_GTK4 "Build the optional GTK4 teaching window" OFF)
add_library(btb-activity STATIC activity.c)
target_include_directories(btb-activity PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(btb-activity PUBLIC Umicom::base)
add_executable(btb-activity-console console.c)
target_link_libraries(btb-activity-console PRIVATE btb-activity)
set(_activity_targets btb-activity btb-activity-console)
if(BUILD_TESTING)
    add_executable(btb-activity-tests test_activity.c)
    target_link_libraries(btb-activity-tests PRIVATE btb-activity)
    list(APPEND _activity_targets btb-activity-tests)
    foreach(_case IN ITEMS first last full invalid null in-place)
        add_test(NAME "bits2banking.framework_activity.${_case}"
            COMMAND btb-activity-tests "${_case}")
        set_tests_properties("bits2banking.framework_activity.${_case}" PROPERTIES TIMEOUT 10)
    endforeach()
endif()
if(BTB_ACTIVITY_GTK4)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
    add_executable(btb-activity-window window.c)
    target_link_libraries(btb-activity-window PRIVATE btb-activity PkgConfig::GTK4)
    list(APPEND _activity_targets btb-activity-window)
endif()
foreach(_target IN LISTS _activity_targets)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
