# Bits2Banking | Umicom Foundation | Independent C23 classroom model, not a production TMS.
cmake_minimum_required(VERSION 3.24)
project(BtbTreasuryCashLadder LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb-treasury-cash-ladder STATIC cash_ladder.c)
target_include_directories(btb-treasury-cash-ladder PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-treasury-cash-demo demo.c)
add_executable(btb-treasury-cash-test test_cash_ladder.c)
target_link_libraries(btb-treasury-cash-demo PRIVATE btb-treasury-cash-ladder)
target_link_libraries(btb-treasury-cash-test PRIVATE btb-treasury-cash-ladder)
foreach(target IN ITEMS btb-treasury-cash-ladder btb-treasury-cash-demo btb-treasury-cash-test)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
 endif()
endforeach()
if(BUILD_TESTING)
 foreach(case IN ITEMS init duplicate currency confirmed settled invalid-date invalid-currency atomic-output overflow capacity leap negative)
  add_test(NAME "bits2banking.treasury.${case}" COMMAND btb-treasury-cash-test "${case}")
 endforeach()
 add_test(NAME bits2banking.treasury.demo COMMAND btb-treasury-cash-demo)
endif()
