# Bits to Banking | Umicom Foundation | Simulated C23 service-dispatch lesson
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingServiceDispatch LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(btb_service_dispatch STATIC dispatch.c)
target_include_directories(btb_service_dispatch PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-service-dispatch main.c)
target_link_libraries(btb-service-dispatch PRIVATE btb_service_dispatch)
if(BUILD_TESTING)
    add_executable(btb-service-dispatch-test test_dispatch.c)
    target_link_libraries(btb-service-dispatch-test PRIVATE btb_service_dispatch)
    add_test(NAME bits2banking.syscall_dispatch COMMAND btb-service-dispatch-test)
    add_test(NAME bits2banking.syscall_dispatch.demo COMMAND btb-service-dispatch)
    set_tests_properties(bits2banking.syscall_dispatch.demo PROPERTIES
        PASS_REGULAR_EXPRESSION "Training events: 1")
endif()
foreach(target IN ITEMS btb_service_dispatch btb-service-dispatch)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(TARGET btb-service-dispatch-test AND NOT MSVC)
    target_compile_options(btb-service-dispatch-test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
