# Bits to Banking | Umicom Foundation | MIT | Hosted C23 scheduling model
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingRoundRobin LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(btb_round_robin STATIC round_robin.c)
target_include_directories(btb_round_robin PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-round-robin main.c)
target_link_libraries(btb-round-robin PRIVATE btb_round_robin)
if(BUILD_TESTING)
    add_executable(btb-round-robin-test test_round_robin.c)
    target_link_libraries(btb-round-robin-test PRIVATE btb_round_robin)
    add_test(NAME bits2banking.scheduler.fairness COMMAND btb-round-robin-test sequence)
    add_test(NAME bits2banking.scheduler.refusals COMMAND btb-round-robin-test refusals)
    set_tests_properties(bits2banking.scheduler.fairness bits2banking.scheduler.refusals
        PROPERTIES LABELS "education;c23;scheduling;kernel-learning")
endif()
foreach(target IN ITEMS btb_round_robin btb-round-robin btb-round-robin-test)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        endif()
    endif()
endforeach()
