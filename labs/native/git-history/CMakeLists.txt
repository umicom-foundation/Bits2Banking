cmake_minimum_required(VERSION 3.24)
project(Bits2BankingGitHistory LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(btb-git-history STATIC graph.c)
target_include_directories(btb-git-history PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-git-history-test test_graph.c)
add_executable(btb-git-history-demo graph_demo.c)
target_link_libraries(btb-git-history-test PRIVATE btb-git-history)
target_link_libraries(btb-git-history-demo PRIVATE btb-git-history)
foreach(target IN ITEMS btb-git-history btb-git-history-test btb-git-history-demo)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endforeach()
if(BUILD_TESTING)
    foreach(case IN ITEMS equal behind ahead diverged merge dirty detached invalid capacity stress)
        add_test(NAME bits2banking.git_history.${case}
                 COMMAND btb-git-history-test "${case}")
    endforeach()
    add_test(NAME bits2banking.git_history.demo COMMAND btb-git-history-demo)
endif()
