cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BtbCompletionQueue LANGUAGES C)
    include(CTest)
endif()
add_library(btb_completion_queue STATIC queue.c)
target_include_directories(btb_completion_queue PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(btb_completion_queue PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(btb_completion_queue PRIVATE /W4)
else()
    target_compile_options(btb_completion_queue PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(btb-completion-queue-demo main.c)
    add_executable(btb-completion-queue-test test_queue.c)
    target_link_libraries(btb-completion-queue-demo PRIVATE btb_completion_queue)
    target_link_libraries(btb-completion-queue-test PRIVATE btb_completion_queue)
    if(BUILD_TESTING)
        add_test(NAME bits2banking.completion_queue COMMAND btb-completion-queue-test)
        add_test(NAME bits2banking.completion_queue.demo COMMAND btb-completion-queue-demo)
    endif()
endif()
