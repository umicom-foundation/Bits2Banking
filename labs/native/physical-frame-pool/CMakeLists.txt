# Bits2Banking — physical 4 KiB frame ownership in C23, no real hardware.
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingPhysicalFrames LANGUAGES C)
include(CTest)
add_library(btb_frame_pool STATIC frame_pool.c)
set_target_properties(btb_frame_pool PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(btb_frame_pool PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-frame-pool-demo main.c)
target_link_libraries(btb-frame-pool-demo PRIVATE btb_frame_pool)
add_executable(btb-frame-pool-test test_frame_pool.c)
target_link_libraries(btb-frame-pool-test PRIVATE btb_frame_pool)
foreach(target IN ITEMS btb_frame_pool btb-frame-pool-demo btb-frame-pool-test)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
if(BUILD_TESTING)
    add_test(NAME bits2banking.frame_pool.ownership COMMAND btb-frame-pool-test)
    add_test(NAME bits2banking.frame_pool.demo COMMAND btb-frame-pool-demo)
    set_tests_properties(bits2banking.frame_pool.ownership bits2banking.frame_pool.demo
        PROPERTIES LABELS "education;c23;physical-memory")
endif()
