# Bits to Banking — Umicom Foundation — MIT
# Build a tiny checked logical timer, without OS sleeps or external scripts.
cmake_minimum_required(VERSION 3.24)
project(Bits2BankingMonotonicDeadlines LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(btb_deadline STATIC deadline.c)
target_include_directories(btb_deadline PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(btb-deadline main.c)
target_link_libraries(btb-deadline PRIVATE btb_deadline)
if(BUILD_TESTING)
    add_executable(btb-deadline-test test_deadline.c)
    target_link_libraries(btb-deadline-test PRIVATE btb_deadline)
    add_test(NAME bits2banking.deadline.boundary COMMAND btb-deadline-test)
    set_tests_properties(bits2banking.deadline.boundary PROPERTIES LABELS "education;c23;clock")
endif()
foreach(target IN ITEMS btb_deadline btb-deadline btb-deadline-test)
    if(TARGET "${target}")
        if(MSVC)
            target_compile_options("${target}" PRIVATE /W4)
        else()
            target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        endif()
    endif()
endforeach()
